// 0: Last Edit Field Position (for power-off resume)

// 8-1467: ALARM1 data (365 days * 2 bytes each = 730 bytes) - FAJAR START
// 1468-2197: ALARM2 data (365 days * 2 bytes each = 730 bytes) - FAJAR END

// 2928-3657: ALARM4 data (365 days * 2 bytes each = 730 bytes) - JOHOR END

// 4388-5117: ALARM6 data (365 days * 2 bytes each = 730 bytes) - ASAR END
// 5118-5847: ALARM7 data (365 days * 2 bytes each = 730 bytes) - MAGRIB START
// 5848-6577: ALARM8 data (365 days * 2 bytes each = 730 bytes) - MAGRIB END

// 7308-8037: ALARM10 data (365 days * 2 bytes each = 730 bytes) - ESHA END

// Function to save last edit field position to EEPROM
void saveLastEditField(int field) {
  writeEEPROM(EEPROM_I2C_ADDRESS, LAST_EDIT_FIELD_ADDR, (byte)field);
}

// Function to read last edit field position from EEPROM
int readLastEditField() {
  byte field = readEEPROM(EEPROM_I2C_ADDRESS, LAST_EDIT_FIELD_ADDR);
  // 31 fields total (0-30): 7 time/date + 24 alarm fields
  if (field == 0xFF || field > 30) return 0; 
  return (int)field;
}

// JUMMA Functions for Arduino Internal EEPROM
void saveJummaToInternalEEPROM() {
  EEPROM.write(INTERNAL_JUMMA_START_HOUR, jummaStartHour);
  EEPROM.write(INTERNAL_JUMMA_START_MINUTE, jummaStartMinute);
  EEPROM.write(INTERNAL_JUMMA_END_HOUR, jummaEndHour);
  EEPROM.write(INTERNAL_JUMMA_END_MINUTE, jummaEndMinute);
}

void loadJummaFromInternalEEPROM() {
  byte startH = EEPROM.read(INTERNAL_JUMMA_START_HOUR);
  byte startM = EEPROM.read(INTERNAL_JUMMA_START_MINUTE);
  byte endH = EEPROM.read(INTERNAL_JUMMA_END_HOUR);
  byte endM = EEPROM.read(INTERNAL_JUMMA_END_MINUTE);
  
  // Validate and set defaults if invalid
  if (startH == 0xFF || startH < 1 || startH > 12) jummaStartHour = 12; else jummaStartHour = startH;
  if (startM == 0xFF || startM > 59) jummaStartMinute = 30; else jummaStartMinute = startM;
  if (endH == 0xFF || endH < 1 || endH > 12) jummaEndHour = 2; else jummaEndHour = endH;
  if (endM == 0xFF || endM > 59) jummaEndMinute = 0; else jummaEndMinute = endM;
}

// EEPROM Functions for 24C512 (Enhanced)
void writeEEPROM(int deviceaddress, unsigned int eeaddress, byte data) {
  Wire.beginTransmission(deviceaddress);
  Wire.write((int)(eeaddress >> 8)); // MSB
  Wire.write((int)(eeaddress & 0xFF)); // LSB
  Wire.write(data);
  Wire.endTransmission();
  delay(10); // 24C512 needs longer delay
}

byte readEEPROM(int deviceaddress, unsigned int eeaddress) {
  byte rdata = 0xFF;
  Wire.beginTransmission(deviceaddress);
  Wire.write((int)(eeaddress >> 8)); // MSB
  Wire.write((int)(eeaddress & 0xFF)); // LSB
  Wire.endTransmission();
  Wire.requestFrom(deviceaddress, 1);
  if (Wire.available()) rdata = Wire.read();
  return rdata;
}

// Function to initialize all 365 days with default values - 10 ALARMS ONLY
void initializeEEPROMAlarms() {
  Serial.println("Initializing 24C512 EEPROM with default alarm values...");
  for (int day = 1; day <= 365; day++) {
    writeDayAlarmToEEPROM(day, 5, 0, 6, 30, 12, 0, 1, 30, 3, 30, 5, 0, 6, 0, 7, 0, 8, 0, 10, 0);
    
    // Show progress every 50 days
    if (day % 50 == 0) {
      Serial.print("Initialized day ");
      Serial.println(day);
    }
  }
  Serial.println("24C512 EEPROM initialization complete!");
}

// Function to verify 24C512
bool verify24C512() {
  Serial.println("Verifying EEPROM type...");
  
  // Write test pattern to high address (only available in 24C512)
  unsigned int testAddress = 32768; // This address doesn't exist in 24C32
  byte testPattern = 0xA5;
  
  writeEEPROM(EEPROM_I2C_ADDRESS, testAddress, testPattern);
  delay(50);
  byte readValue = readEEPROM(EEPROM_I2C_ADDRESS, testAddress);
  
  if (readValue == testPattern) {
    Serial.println("Confirmed: 24C512 EEPROM detected");
    return true;
  } else {
    Serial.println("Warning: May not be 24C512 or connection issue");
    return false;
  }
}

void setup() {
  Serial.begin(9600);
  Wire.begin();
  rtc.begin();

  // Initialize IR receiver
  IrReceiver.begin(IR_RECEIVE_PIN, ENABLE_LED_FEEDBACK);

  pinMode(dataPin, OUTPUT);
  pinMode(clockPin, OUTPUT);
  pinMode(strobePin, OUTPUT);
  pinMode(menuPin, INPUT_PULLUP);
  pinMode(incPin, INPUT_PULLUP);
  pinMode(decPin, INPUT_PULLUP);
  pinMode(leftPin, INPUT_PULLUP);
  pinMode(alarmLEDPin, OUTPUT);
  pinMode(pmLEDPin, OUTPUT);

  // 24C512 EEPROM Test and Verification
  Serial.println("=== 24C512 EEPROM System ===");
  
  // Verify that we have 24C512
  if (verify24C512()) {
    Serial.println("✓ 24C512 verified successfully");
  } else {
    Serial.println("✗ 24C512 verification failed - check connections");
  }

  // Enhanced EEPROM connection test
  Serial.println("Testing 24C512 EEPROM connection...");
  
  // Test low address (compatible with both 24C32 and 24C512)
  writeEEPROM(EEPROM_I2C_ADDRESS, 1, 0xAA);
  delay(50);
  byte testLow = readEEPROM(EEPROM_I2C_ADDRESS, 1);
  
  // Test high address (only available in 24C512)
  writeEEPROM(EEPROM_I2C_ADDRESS, 16384, 0x55);
  delay(50);
  byte testHigh = readEEPROM(EEPROM_I2C_ADDRESS, 16384);
  
  if (testLow == 0xAA && testHigh == 0x55) {
    Serial.println("✓ 24C512 EEPROM connection perfect!");
  } else if (testLow == 0xAA) {
    Serial.println("⚠ EEPROM responding but high address failed");
  } else {
    Serial.println("✗ EEPROM connection failed!");
  }

  // Check if this is first boot (check a signature byte)
  byte signature = readEEPROM(EEPROM_I2C_ADDRESS, 2);
  if (signature != 0xBE) { // 0xBE is our signature for initialized EEPROM
    Serial.println("First boot detected - initializing EEPROM...");
    initializeEEPROMAlarms();
    // Write signature to indicate EEPROM is initialized
    writeEEPROM(EEPROM_I2C_ADDRESS, 2, 0xBE);
    Serial.println("EEPROM initialization complete");
  } else {
    Serial.println("EEPROM already initialized - loading existing data");
  }

  // Load JUMMA from Arduino Internal EEPROM
  loadJummaFromInternalEEPROM();
  Serial.print("JUMMA loaded from internal EEPROM: ");
  Serial.print(jummaStartHour); Serial.print(":"); Serial.print(jummaStartMinute);
  Serial.print(" - "); Serial.print(jummaEndHour); Serial.print(":"); Serial.println(jummaEndMinute);

  if (rtc.lostPower()) {
    Serial.println("RTC lost power - setting time from compile time");
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  }

  loadTime();
  loadCurrentDayAlarm();

  // Read last edit field position from EEPROM (for power-off resume)
  lastSavedEditField = readLastEditField();
  
  Serial.println("=== System Ready ===");
  Serial.print("24C512 Memory used: ");
  Serial.print(ALARM10_START_ADDR + 730);
  Serial.println(" bytes");
  Serial.println("Arduino Internal EEPROM used for JUMMA: 4 bytes");
  
  // Force initial display update
  displayNeedsUpdate = true;
}

void loop() {
  handleSwitches();
  DateTime now = rtc.now();
  int currentSecond = now.second();

  // Check if second has changed
  if (currentSecond != lastSecond) {
    lastSecond = currentSecond;
    displayNeedsUpdate = true;
  }

  if (editField == -1) {
    // Always update time from RTC when not in edit mode
    hour = now.hour();
    minute = now.minute();
    second = currentSecond;
    day = now.day();
    month = now.month();
    year = now.year() % 100;
    dayOfWeek = now.dayOfTheWeek(); // 0=Sunday, 1=Monday, ..., 5=Friday, 6=Saturday

    // Load alarms for current date
    loadCurrentDayAlarm();
  }

  int displayHour = hour % 12;
  if (displayHour == 0) displayHour = 12;

  digitalWrite(pmLEDPin, (hour >= 12) ? HIGH : LOW);

  // Friday Logic: Use JUMMA instead of JOHOR
  bool isFriday = (dayOfWeek == 5);
  
  // Check all exact alarm matches
  bool alarmMatch2 = (hour % 12) == (alarm2Hour % 12) && minute == alarm2Minute;   // FAJAR END
  bool alarmMatch4, alarmMatch6, alarmMatch8, alarmMatch10;
  bool alarmMatchJummaEnd = false;
  
  if (isFriday) {
    // On Friday, check JUMMA END instead of JOHOR END
    alarmMatchJummaEnd = (hour % 12) == (jummaEndHour % 12) && minute == jummaEndMinute;
    alarmMatch4 = false; // Disable JOHOR END on Friday
  } else {
    alarmMatch4 = (hour % 12) == (alarm4Hour % 12) && minute == alarm4Minute;   // JOHOR END
  }
  
  alarmMatch6 = (hour % 12) == (alarm6Hour % 12) && minute == alarm6Minute;   // ASAR END
  alarmMatch8 = (hour % 12) == (alarm8Hour % 12) && minute == alarm8Minute;   // MAGRIB END
  alarmMatch10 = (hour % 12) == (alarm10Hour % 12) && minute == alarm10Minute; // ESHA END
  
  // Check if RTC time is in NAMAZ ranges
  bool timeInFajarRange = isTimeBetweenAlarms(hour, minute, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute);
  bool timeInJohorRange = false;
  bool timeInJummaRange = false;
  
  if (isFriday) {
    timeInJummaRange = isTimeBetweenAlarms(hour, minute, jummaStartHour, jummaStartMinute, jummaEndHour, jummaEndMinute);
  } else {
    timeInJohorRange = isTimeBetweenAlarms(hour, minute, alarm3Hour, alarm3Minute, alarm4Hour, alarm4Minute);
  }
  
  bool timeInAsarRange = isTimeBetweenAlarms(hour, minute, alarm5Hour, alarm5Minute, alarm6Hour, alarm6Minute);
  bool timeInMagribRange = isTimeBetweenAlarms(hour, minute, alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute);
  bool timeInEshaRange = isTimeBetweenAlarms(hour, minute, alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);

  // Determine current namaz range
  int newNamazRange = 0;
  if (timeInFajarRange) newNamazRange = 1;        // FAJAR
  else if (isFriday && timeInJummaRange) newNamazRange = 7;   // JUMMA (special case)
  else if (!isFriday && timeInJohorRange) newNamazRange = 2; // JOHOR (only non-Friday)
  else if (timeInAsarRange) newNamazRange = 3;    // ASAR
  else if (timeInMagribRange) newNamazRange = 4;  // MAGRIB
  else if (timeInEshaRange) newNamazRange = 5;    // ESHA
  else {
    // Find next namaz time
    newNamazRange = getNextNamazRange(hour, minute, isFriday);
  }

  // Check if namaz range has changed or alarm triggered
  if (newNamazRange != currentNamazRange || alarmMatch2 || alarmMatch4 || alarmMatchJummaEnd || alarmMatch6 || alarmMatch8 || alarmMatch10) {
    currentNamazRange = newNamazRange;
    alarmTriggered = true;
    alarmDisplayStart = millis();
    
    // Determine which alarm/range triggered
    if (alarmMatch2) activeAlarm = 2;
    else if (alarmMatch4) activeAlarm = 4;
    else if (alarmMatchJummaEnd) activeAlarm = 11; // JUMMA END
    else if (alarmMatch6) activeAlarm = 6;
    else if (alarmMatch8) activeAlarm = 8;
    else if (alarmMatch10) activeAlarm = 10;
    else activeAlarm = currentNamazRange * 100; // 100=FAJAR, 200=JOHOR, 300=ASAR, 400=MAGRIB, 500=ESHA, 700=JUMMA
    
    alarmTimeDisplayActive = true;
    triggerDay = day;
    displayNeedsUpdate = true;
  }

  // Check if day has changed - if so, deactivate alarm time display
  if (alarmTimeDisplayActive && day != triggerDay) {
    alarmTimeDisplayActive = false;
    triggerDay = 0;
    displayNeedsUpdate = true;
  }

  // LED shows if any namaz time is active
  bool anyNamazActive = (timeInFajarRange || timeInJohorRange || timeInJummaRange || timeInAsarRange || timeInMagribRange || timeInEshaRange || 
                        alarmMatch2 || alarmMatch4 || alarmMatchJummaEnd || alarmMatch6 || alarmMatch8 || alarmMatch10);
  digitalWrite(alarmLEDPin, anyNamazActive ? HIGH : LOW);

  // Handle blinking completion and return to RTC time
  if (editField != -1 && wasBlinking && !blinkState) {
    wasBlinking = false;
    blinkCount++;
    if (blinkCount >= 10) {
      // Save appropriate data based on edit field
      if (editField >= 7 && editField <= 30) {  // All alarm fields
        if (editField >= 11 && editField <= 14) {
          // JUMMA fields (11-14) - save to Arduino Internal EEPROM
          saveJummaToInternalEEPROM();
        } else {
          // Other alarm fields - save to 24C512 EEPROM
          int dayOfYear = getDayOfYear(day, month, 2000 + year);
          writeDayAlarmToEEPROM(dayOfYear, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute,
                               alarm3Hour, alarm3Minute, alarm4Hour, alarm4Minute,
                               alarm5Hour, alarm5Minute, alarm6Hour, alarm6Minute,
                               alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute,
                               alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);
        }
      }
      else if (editField >= 0 && editField <= 6) {
        saveTime();
      }

      // Save current edit field position to EEPROM before exiting edit mode
      lastSavedEditField = editField;
      saveLastEditField(lastSavedEditField);
      
      // Reset edit field and immediately update time from RTC
      editField = -1;
      editModeJustStarted = false;
      
      // Force immediate update from RTC
      DateTime currentTime = rtc.now();
      hour = currentTime.hour();
      minute = currentTime.minute();
      second = currentTime.second();
      day = currentTime.day();
      month = currentTime.month();
      year = currentTime.year() % 100;
      dayOfWeek = currentTime.dayOfTheWeek();

      // Load alarms for current date
      loadCurrentDayAlarm();
      displayNeedsUpdate = true;
    }
  }

  // Handle blinking for edit mode
  if (editField != -1 && millis() - lastBlink > 500) {
    blinkState = !blinkState;
    lastBlink = millis();
    displayNeedsUpdate = true; // Update display when blinking changes
    if (blinkState) wasBlinking = true;
  }

  // Update display only once per second or when needed
  if (displayNeedsUpdate && (millis() - lastDisplayUpdate >= 1000 || editField != -1)) {
    updateDisplay(displayHour);
    lastDisplayUpdate = millis();
    displayNeedsUpdate = false;
  }

  // Reduced delay for better button responsiveness
  delay(10);
}

int getDayOfYear(int day, int month, int year) {
  int daysInMonth[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
  
  // Check for leap year
  if (isLeapYear(year)) {
    daysInMonth[2] = 29;
  }

  int dayOfYear = 0;  
  for (int i = 1; i < month; i++) {
    dayOfYear += daysInMonth[i];
  }
  dayOfYear += day;
  return dayOfYear;
}

bool isLeapYear(int year) {
  return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
}

// Function to check if current time is between two alarms
bool isTimeBetweenAlarms(int currentHour, int currentMinute, int startH, int startM, int endH, int endM) {
  // Convert times to minutes for easier comparison
  int currentTotalMinutes = (currentHour % 12) * 60 + currentMinute;
  int startTotalMinutes = (startH % 12) * 60 + startM;
  int endTotalMinutes = (endH % 12) * 60 + endM;
  
  // Handle case where end time is earlier than start time (crosses midnight)
  if (endTotalMinutes < startTotalMinutes) {
    // Range crosses midnight
    return (currentTotalMinutes >= startTotalMinutes || currentTotalMinutes <= endTotalMinutes);
  } else {
    // Normal range within same day
    return (currentTotalMinutes >= startTotalMinutes && currentTotalMinutes <= endTotalMinutes);
  }
}

// Function to get next namaz range when no current namaz is active - UPDATED FOR FRIDAY
int getNextNamazRange(int currentHour, int currentMinute, bool isFriday) {
  int currentTotalMinutes = currentHour * 60 + currentMinute;
  
  // Convert all namaz times to total minutes
  int fajarStart = convertTo24HourMinutes(alarm1Hour, alarm1Minute, true); // AM
  int johorStart = convertTo24HourMinutes(alarm3Hour, alarm3Minute, false); // PM
  int jummaStart = convertTo24HourMinutes(jummaStartHour, jummaStartMinute, false); // PM
  int asarStart = convertTo24HourMinutes(alarm5Hour, alarm5Minute, false); // PM
  int magribStart = convertTo24HourMinutes(alarm7Hour, alarm7Minute, false); // PM
  int eshaStart = convertTo24HourMinutes(alarm9Hour, alarm9Minute, false); // PM
  
  // Find next namaz based on day
  if (currentTotalMinutes < fajarStart) return 6; // Next is FAJAR
  else if (isFriday && currentTotalMinutes < jummaStart) return 6; // Next is JUMMA
  else if (!isFriday && currentTotalMinutes < johorStart) return 6; // Next is JOHOR
  else if (currentTotalMinutes < asarStart) return 6; // Next is ASAR
  else if (currentTotalMinutes < magribStart) return 6; // Next is MAGRIB
  else if (currentTotalMinutes < eshaStart) return 6; // Next is ESHA
  else return 6; // Next day FAJAR
}

int convertTo24HourMinutes(int hour, int minute, bool isAM) {
  int hour24 = hour;
  if (!isAM && hour != 12) hour24 += 12;
  if (isAM && hour == 12) hour24 = 0;
  return hour24 * 60 + minute;
}

void loadCurrentDayAlarm() {
  int dayOfYear = getDayOfYear(day, month, 2000 + year);

  readDayAlarmFromEEPROM(dayOfYear, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute,

                        alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute,
                        alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);
}

void loadTime() {
  DateTime now = rtc.now();
  hour = now.hour();
  minute = now.minute();
  second = now.second();
  day = now.day();
  month = now.month();
  year = now.year() % 100;
  dayOfWeek = now.dayOfTheWeek();
}

void saveTime() {
  rtc.adjust(DateTime(2000 + year, month, day, hour, minute, second));
}

void handleSwitches() {
  static bool lastMenu = HIGH, lastInc = HIGH, lastDec = HIGH, lastLeft = HIGH;
  bool menuState = digitalRead(menuPin);
  bool incState = digitalRead(incPin);
  bool decState = digitalRead(decPin);
  bool leftState = digitalRead(leftPin);

  unsigned long now = millis();
  static unsigned long lastIncHold = 0, lastDecHold = 0, lastMenuHold = 0, lastLeftHold = 0;

  // Manual Switch Handling - 31 FIELDS (0-30)
  if (lastMenu == HIGH && menuState == LOW) {
    if (editField >= 0 && editField <= 6) saveTime();
    if (editField >= 7 && editField <= 30) {
      if (editField >= 11 && editField <= 14) {
        // JUMMA fields - save to Arduino Internal EEPROM
        saveJummaToInternalEEPROM();
      } else {
        // Other alarm fields - save to 24C512 EEPROM
        int dayOfYear = getDayOfYear(day, month, 2000 + year);
        writeDayAlarmToEEPROM(dayOfYear, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute,
                             alarm3Hour, alarm3Minute, alarm4Hour, alarm4Minute,
                             alarm5Hour, alarm5Minute, alarm6Hour, alarm6Minute,
                             alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute,
                             alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);
      }
    }

    int previousEditField = editField;
    
    // NEW LOGIC: If editField is -1 (not in edit mode), decide where to start
    if (editField == -1) {
      // Check if this is power-on or after blinking completed
      if (lastSavedEditField != -1) {
        // Resume from last saved position (blinking completed, button pressed again)
        editField = lastSavedEditField;
        editModeJustStarted = true;
      } else {
        // Power-on case, start from hour (position 0)
        editField = 0;
        editModeJustStarted = true;
      }
    } else {
      // Normal increment through fields
      editField = (editField + 1) % 31;  // 31 fields (0-30)
    }

    // If exiting edit mode, refresh time from RTC
    if (previousEditField != -1 && editField == -1) {
      DateTime currentTime = rtc.now();
      hour = currentTime.hour();
      minute = currentTime.minute();
      second = currentTime.second();
      day = currentTime.day();
      month = currentTime.month();
      year = currentTime.year() % 100;
      dayOfWeek = currentTime.dayOfTheWeek();
      loadCurrentDayAlarm();
    }

    blinkCount = 0;
    lastMenuHold = now;
    displayNeedsUpdate = true;
  } else if (menuState == LOW && now - lastMenuHold > 500) {
    if (editField >= 0 && editField <= 6) saveTime();
    if (editField >= 7 && editField <= 30) {
      if (editField >= 11 && editField <= 14) {
        saveJummaToInternalEEPROM();
      } else {
        int dayOfYear = getDayOfYear(day, month, 2000 + year);
        writeDayAlarmToEEPROM(dayOfYear, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute,
                             alarm3Hour, alarm3Minute, alarm4Hour, alarm4Minute,
                             alarm5Hour, alarm5Minute, alarm6Hour, alarm6Minute,
                             alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute,
                             alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);
      }
    }

    int previousEditField = editField;
    editField = (editField + 1) % 31;  // 31 fields

    // If exiting edit mode, refresh time from RTC
    if (previousEditField != -1 && editField == -1) {
      DateTime currentTime = rtc.now();
      hour = currentTime.hour();
      minute = currentTime.minute();
      second = currentTime.second();
      day = currentTime.day();
      month = currentTime.month();
      year = currentTime.year() % 100;
      dayOfWeek = currentTime.dayOfTheWeek();
      loadCurrentDayAlarm();
    }

    blinkCount = 0;
    lastMenuHold = now;
    displayNeedsUpdate = true;
  }

  if (lastLeft == HIGH && leftState == LOW) {
    if (editField >= 0 && editField <= 6) saveTime();
    if (editField >= 7 && editField <= 30) {
      if (editField >= 11 && editField <= 14) {
        saveJummaToInternalEEPROM();
      } else {
        int dayOfYear = getDayOfYear(day, month, 2000 + year);
        writeDayAlarmToEEPROM(dayOfYear, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute,
                             alarm3Hour, alarm3Minute, alarm4Hour, alarm4Minute,
                             alarm5Hour, alarm5Minute, alarm6Hour, alarm6Minute,
                             alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute,
                             alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);
      }
    }

    int previousEditField = editField;
    
    // NEW LOGIC: If editField is -1 (not in edit mode), decide where to start
    if (editField == -1) {
      // Check if this is power-on or after blinking completed
      if (lastSavedEditField != -1) {
        // Resume from last saved position (blinking completed, button pressed again)
        editField = lastSavedEditField;
        editModeJustStarted = true;
      } else {
        // Power-on case, start from hour (position 0)
        editField = 0;
        editModeJustStarted = true;
      }
    } else {
      // Normal decrement through fields
      editField = (editField + 30) % 31;  // 31 fields
    }

    // If exiting edit mode, refresh time from RTC
    if (previousEditField != -1 && editField == -1) {
      DateTime currentTime = rtc.now();
      hour = currentTime.hour();
      minute = currentTime.minute();
      second = currentTime.second();
      day = currentTime.day();
      month = currentTime.month();
      year = currentTime.year() % 100;
      dayOfWeek = currentTime.dayOfTheWeek();
      loadCurrentDayAlarm();
    }

    blinkCount = 0;
    lastLeftHold = now;
    displayNeedsUpdate = true;
  } else if (leftState == LOW && now - lastLeftHold > 500) {
    if (editField >= 0 && editField <= 6) saveTime();
    if (editField >= 7 && editField <= 30) {
      if (editField >= 11 && editField <= 14) {
        saveJummaToInternalEEPROM();
      } else {
        int dayOfYear = getDayOfYear(day, month, 2000 + year);
        writeDayAlarmToEEPROM(dayOfYear, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute,
                             alarm3Hour, alarm3Minute, alarm4Hour, alarm4Minute,
                             alarm5Hour, alarm5Minute, alarm6Hour, alarm6Minute,
                             alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute,
                             alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);
      }
    }

    int previousEditField = editField;
    editField = (editField + 30) % 31;  // 31 fields

    // If exiting edit mode, refresh time from RTC
    if (previousEditField != -1 && editField == -1) {
      DateTime currentTime = rtc.now();
      hour = currentTime.hour();
      minute = currentTime.minute();
      second = currentTime.second();
      day = currentTime.day();
      month = currentTime.month();
      year = currentTime.year() % 100;
      dayOfWeek = currentTime.dayOfTheWeek();
      loadCurrentDayAlarm();
    }

    blinkCount = 0;
    lastLeftHold = now;
    displayNeedsUpdate = true;
  }

  if (editField != -1) {
    if (lastInc == HIGH && incState == LOW) {
      incrementField();
      blinkCount = 0;
      lastIncHold = now;
      displayNeedsUpdate = true;
    } else if (incState == LOW && now - lastIncHold > 500) {
      incrementField();
      blinkCount = 0;
      lastIncHold = now;
      displayNeedsUpdate = true;
    }

    if (lastDec == HIGH && decState == LOW) {
      decrementField();
      blinkCount = 0;
      lastDecHold = now;
      displayNeedsUpdate = true;
    } else if (decState == LOW && now - lastDecHold > 500) {
      decrementField();
      blinkCount = 0;
      lastDecHold = now;
      displayNeedsUpdate = true;
    }
  }

  // IR Remote Handling - 31 FIELDS WITH BLINK RESUME LOGIC
  if (IrReceiver.decode()) {
    uint8_t cmd = IrReceiver.decodedIRData.command;
    switch (cmd) {
      case 26: // UP button (MENU/NEXT)
        if (editField >= 0 && editField <= 6) saveTime();
        if (editField >= 7 && editField <= 30) {
          if (editField >= 11 && editField <= 14) {
            saveJummaToInternalEEPROM();
          } else {
            int dayOfYear = getDayOfYear(day, month, 2000 + year);
            writeDayAlarmToEEPROM(dayOfYear, alarm1Hour, alarm1Minute, alarm2Hour, alarm2Minute,
                                 alarm3Hour, alarm3Minute, alarm4Hour, alarm4Minute,
                                 alarm5Hour, alarm5Minute, alarm6Hour, alarm6Minute,
                                 alarm7Hour, alarm7Minute, alarm8Hour, alarm8Minute,
                                 alarm9Hour, alarm9Minute, alarm10Hour, alarm10Minute);
          }
        }
        {
          int previousEditField = editField;
          
          // NEW LOGIC: If editField is -1 (not in edit mode), decide where to start
          if (editField == -1) {
            // Check if this is power-on or after blinking completed
            if (lastSavedEditField != -1) {
              // Resume from last saved position (blinking completed, button pressed again)
              editField = lastSavedEditField;
            } else {
              // Power-on case, start from hour (position 0)
              editField = 0;
            }
          } else {
            // Normal increment through fields
            editField = (editField + 1) % 31;
            // If we've completed the cycle (reached 0 again), exit edit mode
            if (editField == 0 && previousEditField == 30) {
              lastSavedEditField = previousEditField;  // Save current position
              //editField = -1;  // Exit edit mode   for right scroll ok
              DateTime currentTime = rtc.now();
              hour = currentTime.hour();
              minute = currentTime.minute();
              second = currentTime.second();
              day = currentTime.day();
              month = currentTime.month();
              year = currentTime.year() % 100;
              dayOfWeek = currentTime.dayOfTheWeek();
              loadCurrentDayAlarm();
            }
          }
        }
        blinkCount = 0;
        displayNeedsUpdate = true;
        break;

      case 27: // DOWN button (Increment current field)
        if (editField != -1) {
          incrementField();
          blinkCount = 0;
          displayNeedsUpdate = true;
        }
        break;

      case 30: // LEFT button (Move to previous field)
        {
          if (editField == -1) {
            if (lastSavedEditField != -1) {
              editField = lastSavedEditField;
            } else {
              editField = 0;
            }
          } else {
            // Normal decrement with cycling
            editField = (editField + 30) % 31;
          }
        }
        blinkCount = 0;
        displayNeedsUpdate = true;
        break;
      case 31: // RIGHT button (Decrement current field)
        if (editField != -1) {
          decrementField();
          blinkCount = 0;
          displayNeedsUpdate = true;
        }
        break;
    }
    IrReceiver.resume(); // Always call after decode
  }

  lastMenu = menuState;
  lastInc = incState;
  lastDec = decState;
  lastLeft = leftState;
}

void incrementField() {
  if (editField == 0) hour = (hour + 1) % 24;
  if (editField == 1) minute = (minute + 1) % 60;
  if (editField == 2) second = (second + 1) % 60;
  if (editField == 3) {
    day = (day < 31) ? day + 1 : 1;
    loadCurrentDayAlarm(); // Load alarms for new date
  }
  if (editField == 4) {
    month = (month < 12) ? month + 1 : 1;
    loadCurrentDayAlarm(); // Load alarms for new date
  }
  if (editField == 5) {
    year = (year < 99) ? year + 1 : 0;
    loadCurrentDayAlarm(); // Load alarms for new date
  }
  if (editField == 6) {
    dayOfWeek = (dayOfWeek + 1) % 7; // Day of week (0-6)
  }
  
  // NEW FIELD ORDER: ফজর→জুম্মা→যোহর→আসর→মাগরিব→এশা
  // FAJAR START/END (fields 7-10)
  if (editField == 7) alarm1Hour = (alarm1Hour % 12) + 1;
  if (editField == 8) alarm1Minute = (alarm1Minute + 1) % 60;
  if (editField == 9) alarm2Hour = (alarm2Hour % 12) + 1;
  if (editField == 10) alarm2Minute = (alarm2Minute + 1) % 60;
  
  // JUMMA START/END (fields 11-14)
  if (editField == 11) jummaStartHour = (jummaStartHour % 12) + 1;
  if (editField == 12) jummaStartMinute = (jummaStartMinute + 1) % 60;
  if (editField == 13) jummaEndHour = (jummaEndHour % 12) + 1;
  if (editField == 14) jummaEndMinute = (jummaEndMinute + 1) % 60;
  
  // JOHAR START/END (fields 15-18)
  if (editField == 15) alarm3Hour = (alarm3Hour % 12) + 1;
  if (editField == 16) alarm3Minute = (alarm3Minute + 1) % 60;
  if (editField == 17) alarm4Hour = (alarm4Hour % 12) + 1;
  if (editField == 18) alarm4Minute = (alarm4Minute + 1) % 60;
  
  // ASAR START/END (fields 19-22)
  if (editField == 19) alarm5Hour = (alarm5Hour % 12) + 1;
  if (editField == 20) alarm5Minute = (alarm5Minute + 1) % 60;
  if (editField == 21) alarm6Hour = (alarm6Hour % 12) + 1;
  if (editField == 22) alarm6Minute = (alarm6Minute + 1) % 60;
  
  // MAGRIB START/END (fields 23-26)
  if (editField == 23) alarm7Hour = (alarm7Hour % 12) + 1;
  if (editField == 24) alarm7Minute = (alarm7Minute + 1) % 60;
  if (editField == 25) alarm8Hour = (alarm8Hour % 12) + 1;
  if (editField == 26) alarm8Minute = (alarm8Minute + 1) % 60;
  
  // ESHA START/END (fields 27-30)
  if (editField == 27) alarm9Hour = (alarm9Hour % 12) + 1;
  if (editField == 28) alarm9Minute = (alarm9Minute + 1) % 60;
  if (editField == 29) alarm10Hour = (alarm10Hour % 12) + 1;
  if (editField == 30) alarm10Minute = (alarm10Minute + 1) % 60;
}
